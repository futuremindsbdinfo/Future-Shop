export function ProductCard({ id, name, slug, price, salePrice, image, vendorName, inStock }: ProductCardProps) {
  const displayPrice = salePrice ?? price
  const hasDiscount = salePrice && salePrice < price
  const discountPct = hasDiscount
    ? Math.round((1 - salePrice / price) * 100)
    : 0

  return (
    <div className="group relative bg-white rounded-xl border border-border
                    hover:shadow-lg transition-all duration-200 overflow-hidden">
      
      {/* Discount Badge */}
      {hasDiscount && (
        <Badge className="absolute top-2 left-2 z-10 bg-red-500 text-white">

        </Badge>
      )}

      {/* Product Image */}
      <Link href={`/products/${slug}`}>
        <div className="relative h-48 bg-muted overflow-hidden">
          <Image
            src={image}
            alt={name}           {/* Always set alt for SEO + a11y */}
            fill
            className="object-cover group-hover:scale-105 transition-transform duration-300"
            sizes="(max-width: 768px) 50vw, 25vw"
          />
        </div>
      </Link>

      {/* Product Info */}

API Client (Axios with Security)

TypeScript — lib/api.ts
// lib/api.ts — Secure Axios Instance
import axios from 'axios'
import { getToken, clearToken } from './auth'

const api = axios.create({
  baseURL: process.env.NEXT_PUBLIC_API_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json',
    'Accept': 'application/json',
    'X-Requested-With': 'XMLHttpRequest', // CSRF protection signal
  },
  withCredentials: true, // Send cookies for SameSite CSRF protection
})

// Attach Bearer token to every request
api.interceptors.request.use((config) => {
  const token = getToken() // Read from memory, NOT localStorage
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// Handle 401 — token expired → redirect to login
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      clearToken()
      window.location.replace('/login')
    }
    return Promise.reject(error)
  }
)

export default api

Route Protection Middleware

TypeScript — middleware.ts
// middleware.ts — Next.js Route Protection
import { NextResponse } from 'next/server'
import type { NextRequest } from 'next/server'

export function middleware(request: NextRequest) {
  const token = request.cookies.get('lb_token')?.value
  const role = request.cookies.get('lb_role')?.value
  const { pathname } = request.nextUrl

  // Admin routes — must be admin role
  if (pathname.startsWith('/admin')) {
    if (!token || role !== 'admin') {
      return NextResponse.redirect(new URL('/login', request.url))
    }
  }

  // Delivery routes — must be delivery role
  if (pathname.startsWith('/delivery')) {
    if (!token || role !== 'delivery') {
      return NextResponse.redirect(new URL('/login', request.url))
    }
  }

  // Protected customer routes
  if (['/checkout', '/orders'].some(p => pathname.startsWith(p))) {
    if (!token) {
      return NextResponse.redirect(new URL(`/login?redirect=${pathname}`, request.url))
    }
  }

  return NextResponse.next()
}

export const config = {
  matcher: ['/admin/:path*', '/delivery/:path*', '/checkout', '/orders/:path*']
}

Security Architecture

🔐

Security First. Every layer has its own defense. Even if one layer is breached, others must hold. This is defense-in-depth for a financial transaction platform.

Laravel Security Implementation

PHP — OTP Service (Redis-backed, Brute-force Protected)
// app/Services/OtpService.php
class OtpService {
  private const OTP_TTL        = 300;   // 5 minutes
  private const MAX_ATTEMPTS   = 3;
  private const LOCKOUT_TTL    = 3600;  // 1 hour lockout after 3 fails
  private const SEND_LIMIT     = 3;     // Max 3 OTP sends/hour

  public function send(string $phone): void {
    // Check send rate limit
    $sendKey = "otp:send:{$phone}";
    $sends = Redis::incr($sendKey);
    if ($sends === 1) Redis::expire($sendKey, 3600);
    if ($sends > self::SEND_LIMIT) {
      throw new TooManyRequestsException('OTP limit exceeded. Try in 1 hour.');
    }

    // Generate cryptographically secure 6-digit OTP
    $otp = (string) random_int(100000, 999999);

    // Store hashed OTP in Redis (not plain text)
    $hashedOtp = hash_hmac('sha256', $otp, config('app.key'));
    Redis::setex("otp:verify:{$phone}", self::OTP_TTL, $hashedOtp);

    // Send via SSL Wireless
    app(SmsService::class)->send($phone, "Future Shop OTP: {$otp} (expires 5 min)");
  }

  public function verify(string $phone, string $otp): bool {
    // Check if locked out
    $lockKey = "otp:lock:{$phone}";
    if (Redis::exists($lockKey)) {
      throw new AccountLockedException('Too many attempts. Try later.');
    }

    $stored = Redis::get("otp:verify:{$phone}");
    if (!$stored) return false; // expired

    $inputHash = hash_hmac('sha256', $otp, config('app.key'));

    // Constant-time comparison (prevents timing attacks)
    if (!hash_equals($stored, $inputHash)) {
      $attKey = "otp:attempts:{$phone}";
      $attempts = Redis::incr($attKey);
      if ($attempts === 1) Redis::expire($attKey, self::LOCKOUT_TTL);
      if ($attempts >= self::MAX_ATTEMPTS) {
        Redis::setex($lockKey, self::LOCKOUT_TTL, 1); // Lock account
      }
      return false;
    }

    // Invalidate OTP immediately after successful use
    Redis::del("otp:verify:{$phone}", "otp:attempts:{$phone}");
    return true;
  }
}

Payment Webhook Verification

PHP — Payment Webhook (HMAC + Amount Verification)
// app/Http/Controllers/Api/V1/PaymentController.php
public function webhookSuccess(Request $request): JsonResponse {

  // STEP 1: Verify SSL Commerz signature FIRST
  $received   = $request->input('verify_sign');
  $storePass  = config('services.sslcommerz.store_pass');
  $data       = $request->except(['verify_sign', 'verify_key']);
  ksort($data);
  $expected = md5(implode('', array_values($data)) . $storePass);

  if (!hash_equals($expected, $received)) {
    Log::warning('Invalid payment webhook signature', ['ip' => $request->ip()]);
    return response()->json(['error' => 'Invalid signature'], 403);
  }

  // STEP 2: Find order in DB (use tran_id, not client-sent order_id)
  $order = Order::where('order_number', $request->input('tran_id'))->firstOrFail();

  // STEP 3: Re-verify amount matches DB (CRITICAL — never trust client price)
  $paidAmount   = (float) $request->input('amount');
  $expectedAmt  = (float) $order->total;
  if (abs($paidAmount - $expectedAmt) > 0.01) {
    Log::alert('Payment amount mismatch!', compact('paidAmount', 'expectedAmt'));
    return response()->json(['error' => 'Amount mismatch'], 422);
  }

  // STEP 4: Idempotency check (prevent duplicate processing)
  if ($order->payment_status === 'paid') {
    return response()->json(['status' => 'already_processed']);
  }

  // STEP 5: Update order status + log transaction
  DB::transaction(function() use ($order, $request, $paidAmount) {
    $order->update([
      'payment_status' => 'paid',
      'order_status'   => 'confirmed',
    ]);
    Transaction::create([
      'order_id'        => $order->id,
      'gateway'         => 'sslcommerz',
      'gateway_txn_id'  => $request->input('bank_tran_id'),
      'amount'          => $paidAmount,
      'status'          => 'success',
      'raw_response'    => $request->all(),
    ]);
  });

  // STEP 6: Trigger n8n webhook for SMS notification
  dispatch(new SendOrderConfirmationJob($order));
  return response()->json(['status' => 'success']);
}
